│   ├── communication_disorder.c# Eseguibile guasto cassa (Versione Completa)
│   └── user_injector.c         # Eseguibile iniettore utenti (Versione Completa)
│
├── config/                     # File di configurazione
│   ├── standard.conf           # Parametri simulazione standard
│   ├── config_timeout.conf     # Configurazione per terminazione da TIMEOUT
│   ├── config_overload.conf    # Configurazione per terminazione da OVERLOAD
│   └── menu.txt                # Menu offerto dalla mensa
│
└── bin/                        # Cartella creata da make per gli eseguibili
